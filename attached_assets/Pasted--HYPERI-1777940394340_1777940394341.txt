from fastapi import APIRouter, UploadFile, File, Depends, BackgroundTasks, HTTPException

# ── REQUEST/RESPONSE SCHEMAS ──────────────────────────────────────────────────
class CreditAnalysisRequest(BaseModel):
    client_id:          Optional[str]   = Field(None, description="Existing client ID")
    client_name:        str             = Field(..., min_length=2, max_length=255)
    client_email:       str             = Field(..., pattern=r'^[^@]+@[^@]+\.[^@]+$')
    client_phone:       Optional[str]   = None
    annual_income:      Optional[float] = Field(None, ge=0)
    monthly_expenses:   Optional[float] = Field(None, ge=0)
    goals:              Optional[dict]  = None
    report_type:        str             = Field("full", pattern=r'^(full|summary|dispute_only)$')
    priority_focus:     Optional[str]   = Field(None, description="score_boost|debt_reduction|mortgage_prep")

class CreditAnalysisResponse(BaseModel):
    job_id:             str

class ReportStatusResponse(BaseModel):
    job_id:             str
    status:             str             # queued|processing|complete|failed
    progress_pct:       int
    current_layer:      Optional[str]   = None
    report_url:         Optional[str]   = None
    dispute_package_url: Optional[str]  = None
    error_message:      Optional[str]   = None

# ── ROUTES ────────────────────────────────────────────────────────────────────
@router.post("/analyze",
    response_model=CreditAnalysisResponse,
    summary="Submit credit report files for full analysis",
    description="Upload 1-3 bureau credit reports (PDF/HTML/JSON). Returns job ID for async processing."
)
async def analyze_credit(
    request_data: CreditAnalysisRequest,
    equifax_file:    Optional[UploadFile] = File(None),
    transunion_file: Optional[UploadFile] = File(None),
    experian_file:   Optional[UploadFile] = File(None),
    background_tasks: BackgroundTasks = BackgroundTasks(),
    current_user: dict = Depends(verify_jwt_token)
):
    """
    Accepts credit bureau report files, enqueues Hyperion crew analysis,
    returns job ID for polling. Uses Celery for async processing.
    """
    if not any([equifax_file, transunion_file, experian_file]):
        raise HTTPException(status_code=422, detail="At least one bureau file required")

    job_id = str(uuid.uuid4())
    client_id = request_data.client_id or str(uuid.uuid4())

    # Enqueue to Celery
    background_tasks.add_task(
        run_hyperion_analysis,
        job_id=job_id,
        client_id=client_id,
        request_data=request_data.model_dump(),
        files={
            "equifax": equifax_file,
            "transunion": transunion_file,
            "experian": experian_file
        }
    )

    return CreditAnalysisResponse(
        job_id=job_id,
        client_id=client_id,
        status="queued",
        estimated_minutes=8
    )

@router.get("/status/{job_id}",
    response_model=ReportStatusResponse,
    summary="Poll analysis job status"
